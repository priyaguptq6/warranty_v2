# ─ FIREBASE CONFIGURATION (Cloud Backup) ─────────────────────────────
# Setup: https://firebase.google.com/
# 1. Create a Firebase project at https://console.firebase.google.com/
# 2. Go to Project Settings > Service Accounts
# 3. Click "Generate New Private Key"
# 4. Copy the JSON content and set FIREBASE_SERVICE_ACCOUNT_JSON environment variable
# 
# For local development:
#   a. Copy your service account JSON file to: ./service-account-key.json
#   b. Set: FIREBASE_SERVICE_ACCOUNT=/full/path/to/service-account-key.json
#
# For Heroku:
#   a. Paste the entire JSON as single env var (no line breaks)
#   b. Set: heroku config:set FIREBASE_SERVICE_ACCOUNT_JSON='{"type":"service_account"...}'
#
# Note: Database syncs automatically if Firebase is enabled
# FIREBASE_SERVICE_ACCOUNT_JSON=(set via environment variable)

# ─ ANTHROPIC API KEY (AI Photo Analysis) ──────────────────────────
# Setup: https://console.anthropic.com/
# 1. Create/login to your Anthropic account
# 2. Get API key from Dashboard
# 3. Set environment variable: ANTHROPIC_API_KEY=sk-ant-xxx
#
# Features enabled with AI:
#   - Automatic device brand/model/serial extraction from photos
#   - Damage severity assessment  
#   - Repair cost suggestions
#
# Optional: Works fine without it, just disable AI features in admin panel
ANTHROPIC_API_KEY=your_api_key_here

# ─ GMAIL APP PASSWORD (Email Notifications) ──────────────────────
# Setup: Gmail > Account > Security > 2-Step Verification > App Passwords
# 1. Go to: https://myaccount.google.com/apppasswords
# 2. Select "Mail" and "Windows Computer" (or your device)
# 3. Copy the 16-character password
# 4. Set environment: SMTP_USER=yourname@gmail.com, SMTP_PASSWORD=xxxx xxxx xxxx xxxx
#
# Features: Customer status updates, payment reminders, delivery notifications
SMTP_USER=your_gmail@gmail.com
SMTP_PASSWORD=your_16_char_app_password
